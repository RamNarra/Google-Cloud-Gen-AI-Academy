#include <stdio.h>

#include <stdlib.h>

struct node {
int data;
struct node * next;
};

struct node * head = NULL;

// Create Circular Linked List
void create() {
int val;
struct node * temp, * newnode;

printf("Enter zero to stop: ");
while (1) {
scanf("%d", & val);
if (val == 0)
break;

newnode = (struct node * ) malloc(sizeof(struct node));
newnode -> data = val;

if (head == NULL) {
head = newnode;
newnode -> next = head;
} else {
temp = head;
while (temp -> next != head)
temp = temp -> next;

temp -> next = newnode;
newnode -> next = head;
}
}
}

// Traverse
void display() {
struct node * temp;

printf("Contents of linked list are:\n");

if (head == NULL)
return;

temp = head;
do {
printf(" %d", temp -> data);
temp = temp -> next;
} while (temp != head);

printf("\n");
}

// Insert at position
void insert_pos(int pos, int val) {
struct node * temp, * newnode;
int i;

if (pos <= 0) {
printf("Invalid position!\n");
return;
}

if (head == NULL && pos != 1) {
printf("List is empty!\n");
return;
}

newnode = (struct node * ) malloc(sizeof(struct node));
newnode -> data = val;

if (pos == 1) {
if (head == NULL) {
head = newnode;
newnode -> next = head;
} else {
temp = head;
while (temp -> next != head)
temp = temp -> next;

newnode -> next = head;
temp -> next = newnode;
head = newnode;
}
printf("%d is inserted\n", val);
return;
}

temp = head;
for (i = 1; i < pos - 1; i++) {
temp = temp -> next;
if (temp == head) {
printf("Insufficient nodes, %d not inserted.\n", val);
free(newnode);
return;
}
}

newnode -> next = temp -> next;
temp -> next = newnode;
printf("Inserted at position %d!\n", pos);
}

// Delete at position
void delete_pos(int pos) {
struct node * temp, * prev;
int i;

if (pos <= 0) {
printf("Invalid position!\n");
return;
}

if (head == NULL) {
printf("List is empty!\n");
return;
}

temp = head;

if (pos == 1) {
if (head -> next == head) {
printf("%d deleted.\n", head -> data);
free(head);
head = NULL;
return;
}

while (temp -> next != head)
temp = temp -> next;

prev = head;
temp -> next = head -> next;
head = head -> next;

printf("%d deleted.\n", prev -> data);
free(prev);
return;
}

prev = head;
temp = head -> next;

for (i = 2; i < pos; i++) {
prev = temp;
temp = temp -> next;

if (temp == head) {
printf("Insufficient nodes\n");
return;
}
}

prev -> next = temp -> next;
printf("%d deleted.\n", temp -> data);
free(temp);
}

int main() {
int pos, val;

create();
display();

printf("Enter position to insert: ");
scanf("%d", & pos);
printf("Enter value: ");
scanf("%d", & val);

insert_pos(pos, val);
display();

printf("Enter position to be deleted: ");
scanf("%d", & pos);

delete_pos(pos);
display();

return 0;
}

{

{

else
printf("Stack is overflow.\n");
}

{

};

void enqueue(int element){
Q temp=NULL;
temp=(Q)malloc(sizeof(struct queue));
if(temp==NULL){
printf("Queue is overflow.\n");
}
else{
temp->data=element;
temp->next=NULL;
if(front==NULL){
front=temp;
}
else{
rear->next=temp;
}
rear=temp;
printf("Successfully inserted.\n");
}
}
void dequeue(){
Q temp=NULL;
if(front==NULL){
printf("Queue is underflow.\n");
}
else{
temp=front;
if(front==rear){
front=rear=NULL;
}
else{
front=front->next;
}
printf("Deleted value = %d\n",temp->data);
free(temp);

// Using counting sort to sort the elements based on significant places
void countingSort(int array[], int n, int place) {
int *output = (int *)malloc(n * sizeof(int));
int count[10] = {0};
// Calculate count of elements
for (int i = 0; i < n; i++) {
int index = (array[i] / place) % 10;
count[index]++;
}
// Calculate cumulative count
for (int i = 1; i < 10; i++) {
count[i] += count[i - 1];
}
// Place the elements in sorted order
for (int i = n - 1; i >= 0; i--) {
int index = (array[i] / place) % 10;
output[count[index] - 1] = array[i];
count[index]--;
}
// Copy the sorted elements into original array
for (int i = 0; i < n; i++) {
array[i] = output[i];
}
free(output);
}
// Main function to implement radix sort
void radixSort(int array[], int n) {
// Get maximum element
int maxElement = getMax(array, n);
// Apply counting sort to sort elements based on place value
for (int place = 1; maxElement / place > 0; place *= 10) {
countingSort(array, n, place);
}
}

void shellSort(int array[], int n) {
for (int interval = n / 2; interval > 0; interval /= 2) {
for (int i = interval; i < n; i += 1) {
int temp = array[i];
int j;

array[j] = temp;
}
}

/* ---------- BST Node ---------- */
struct node {
int data;
struct node *left, *right;
};

/* ---------- Stack Node (Linked List) ---------- */
struct stack {
struct node *data;
struct stack *next;
};

struct stack *top = NULL;

/* ---------- Stack Operations ---------- */
void push(struct node *x) {
struct stack *temp = (struct stack *)malloc(sizeof(struct stack));
temp->data = x;
temp->next = top;
top = temp;
}

struct node* pop() {
if (top == NULL)
return NULL;

struct stack *temp = top;
struct node *x = temp->data;
top = temp->next;
free(temp);
return x;
}

int isEmpty() {
return top == NULL;
}

/* ---------- BST Operations ---------- */
struct node* createNode(int x) {
struct node *temp = (struct node *)malloc(sizeof(struct node));
temp->data = x;
temp->left = temp->right = NULL;
return temp;
}

struct node* insert(struct node *root, int x) {
struct node *cur = root, *parent = NULL;

while (cur != NULL) {
parent = cur;
if (x < cur->data)
cur = cur->left;
else
cur = cur->right;
}

if (parent == NULL)
return createNode(x);

if (x < parent->data)
parent->left = createNode(x);
else
parent->right = createNode(x);

return root;
}

/* ---------- Inorder Traversal (Non-Recursive) ---------- */
void inorder(struct node *root) {
struct node *cur = root;

while (cur != NULL || !isEmpty()) {
while (cur != NULL) {
push(cur);
cur = cur->left;
}
cur = pop();
printf("%d ", cur->data);
cur = cur->right;
}
}

/* ---------- Preorder Traversal (Non-Recursive) ---------- */
void preorder(struct node *root) {
if (root == NULL) return;

push(root);
while (!isEmpty()) {
struct node *cur = pop();
printf("%d ", cur->data);

if (cur->right) push(cur->right);
if (cur->left)  push(cur->left);
}
}

/* ---------- Postorder Traversal (Non-Recursive) ---------- */
void postorder(struct node *root) {
if (root == NULL) return;

struct stack *top2 = NULL;
struct node *cur;

push(root);
while (!isEmpty()) {
cur = pop();

struct stack *temp = (struct stack *)malloc(sizeof(struct stack));
temp->data = cur;
temp->next = top2;
top2 = temp;

if (cur->left)  push(cur->left);
if (cur->right) push(cur->right);
}

top2 = top2->next;

}
}
